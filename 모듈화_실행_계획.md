   - 팩토리 메서드, 비즈니스 규칙 추가

   ```python
   # core/domain/entities/report.py
   from dataclasses import dataclass
   from datetime import datetime
   import uuid

   @dataclass
   class Report:
       id: str
       created_at: datetime
       observations: List[str]
       emotions: List[str]
       status: str

       @classmethod
       def create_new(cls) -> "Report":
           return cls(
               id=str(uuid.uuid4()),
               created_at=datetime.now(),
               observations=[],
               emotions=[],
               status="draft"
           )

       def add_observation(self, observation: str) -> None:
           if not observation.strip():
               raise ValueError("관찰은 비어있을 수 없습니다")
           self.observations.append(observation)

       def mark_completed(self) -> None:
           if not self.observations or not self.emotions:
               raise ValueError("완료하려면 관찰과 감정이 필요합니다")
           self.status = "completed"
   ```

   ```python
   # presentation/api/routes/analysis.py (100줄)
   from fastapi import APIRouter, Depends
   from features.image_analysis import api as image_analysis

   router = APIRouter(prefix="/api/analysis", tags=["analysis"])

   @router.post("/image")
   async def analyze_image(
       image: UploadFile,
       service = Depends(get_image_analysis_service)
   ):
       # 10줄로 축소 (로직은 feature로 위임)
       image_bytes = await image.read()
       result = image_analysis.analyze_image(image_bytes)
       return {"result": result}
   ```

   ```python
   # presentation/api/dependencies.py
   from features.image_analysis.service import ImageAnalysisService

   from infrastructure.ai_providers.openai_provider import OpenAIProvider

   # 싱글톤 인스턴스
   _openai_provider = None
   _report_repository = None

   def get_openai_provider() -> OpenAIProvider:
       global _openai_provider
       if _openai_provider is None:
   ```

### 2. 첫 번째 데코레이터 (AOP)

```python
# shared/decorators/logged.py
import logging
from functools import wraps
import time

logger = logging.getLogger(__name__)

def log_execution(func):
    """함수 실행을 자동으로 로깅"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        func_name = f"{func.__module__}.{func.__name__}"
        logger.info(f"[START] {func_name}")

        start_time = time.time()
        try:
            result = func(*args, **kwargs)
            execution_time = time.time() - start_time
            logger.info(f"[SUCCESS] {func_name} (took {execution_time:.2f}s)")
            return result
        except Exception as e:
            execution_time = time.time() - start_time
            logger.error(f"[ERROR] {func_name} failed: {e}")
            raise

    return wrapper
```
